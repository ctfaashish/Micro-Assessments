Private Sub btnCalc_Click(sender As Object, e As EventArgs) Handles btnCalc.Click
Dim str1stpricecontrol As String
Dim dblPrice As Double
Dim dblTotaldue As Double
lblTotaldue.Text = String.Empty
lstPrices.Items.Clear()
str1stpricecontrol = InputBox("Enter item's PRICE(0 to end):", "")
Double.TryParse(str1stpricecontrol.ToString, dblPrice)
Do While str1stpricecontrol <> String.Empty
lstPrices.Items.Add(dblPrice)
dblTotaldue += dblPrice
str1stpricecontrol = InputBox("Enter item's PRICE(0 to end):", "")
Double.TryParse(str1stpricecontrol.ToString, dblPrice)
Loop
lblTotaldue.Text = (dblTotaldue * 1.05).ToString("C2")
End Sub
Private Sub btnExit_Click(sender As Object, e As EventArgs) Handles btnExit.Click
Me.Close()
End Sub
End Class
