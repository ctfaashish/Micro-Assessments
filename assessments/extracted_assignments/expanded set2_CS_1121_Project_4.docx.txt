'concert solution: calculates the total number of tickets and total price
'2/15/2017
Private Sub btnCalc_Click(sender As Object, e As EventArgs) Handles btnCalc.Click
'declare variables and constants
Const decBoxticket_PRICE As Decimal = 97.5
Const decPavilionticket_PRICE As Decimal = 55.5
Const decLawnticket_PRICE As Decimal = 21
Const decdiscountbox As Decimal = 0.1
Dim intboxTextbox As Integer
Dim intpavilionTextbox As Integer
Dim intlawnTextbox As Integer
Dim intTotaltickets As Integer
Dim decTotaldue As Decimal
Dim decITEM_PRICE As Decimal
'calculate the total number of tickets sold
Integer.TryParse(boxTextbox.Text, intboxTextbox)
Integer.TryParse(pavilionTextbox.Text, intpavilionTextbox)
Integer.TryParse(lawnTextbox.Text, intlawnTextbox)
'calculate the total amount of tickets and total due
intTotaltickets = intboxTextbox + intpavilionTextbox + intlawnTextbox
decITEM_PRICE = (decLawnticket_PRICE * intlawnTextbox) +
(decPavilionticket_PRICE * intpavilionTextbox) + (decBoxticket_PRICE * intboxTextbox)
decTotaldue = decITEM_PRICE
If discountbox.Checked = True Then
decTotaldue = decTotaldue - (decITEM_PRICE * decdiscountbox)
End If
'display total tickets and amount due
lblTotaltickets.Text = Convert.ToString(intTotaltickets)
lblTotaldue.Text = decTotaldue.ToString("C2")
End Sub
